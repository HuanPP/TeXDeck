\title{光合作用}

\begin{document}

% [GUIDE] A sheet = one topic. The title is the topic itself, centred by
% [GUIDE] default; keep the default for every sheet. \src{N} ends the line of
% [GUIDE] each sheet, \notehead and \notesub (rule 13).
\begin{notesheet}{光合作用的条件与过程}\src{1}

% [GUIDE] Lead-in + parallel bullets: the lead names what is being listed.
\notelead{光合作用需要的三样东西}
\begin{itemize}
    \item \key{光}：主要利用红光和蓝紫光
    \item 二氧化碳：经叶片的\key{气孔}进入
    \item 水：由根吸收，经导管运到叶片
\end{itemize}
% [GUIDE] BAD:  \item 光是光合作用中非常重要的一个条件
% [GUIDE] GOOD: \item 光：主要利用红光和蓝紫光      (says WHAT, not THAT)

% [GUIDE] \notehead splits the sheet; \notesub only when a part has 2+ groups.
\notehead{两个阶段}\src{2-3}
\notesub{光反应（类囊体膜上）}\src{2}
\begin{itemize}
    \item 水被分解，释放\key{氧气}
    \item 光能转化为 ATP 和 NADPH 中的化学能
\end{itemize}
\notesub{暗反应（叶绿体基质中）}\src{3}
\begin{itemize}
    \item 又称\key{卡尔文循环}；不直接需要光，但要用光反应的产物
    \item CO$_2$ 先与 RuBP 结合，再被还原成糖
    \begin{itemize}
        \item 每固定 6 个 CO$_2$，得到 1 个葡萄糖
        \item 消耗 18 个 ATP、12 个 NADPH
    \end{itemize}
\end{itemize}

% [GUIDE] A cause-and-effect chain: one link per bullet, "A → B".
\notehead{光照不足时发生什么}\src{4}
\begin{itemize}
    \item 光照减弱 → 光反应产生的 ATP、NADPH 减少
    \item ATP、NADPH 减少 → 暗反应中 C$_3$ 的还原变慢，C$_3$ 积累
\end{itemize}
\end{notesheet}

\begin{notesheet}{影响光合速率的因素}\src{5}

% [GUIDE] Space follows weight: the material develops light the most, so it
% [GUIDE] gets the most bullets; do not pad the others to match.
\notelead{三个主要因素}
\begin{itemize}
    \item 光照强度：达到\key{光饱和点}之前，越强越快 \aside{之后不再增加}
    \item 温度：通过酶的活性起作用，过高反而下降
    \item CO$_2$ 浓度：温室中常人工补充 \eg{燃烧沼气}
\end{itemize}

% [GUIDE] Label/value pairs: notefacts. Use it for an experiment's set-up,
% [GUIDE] a person's dates, a definition list, settings and quantities.
\notehead{实验：光照强度的影响}\src{6}
\begin{notefacts}
    \fact{材料：}{金鱼藻，放在盛清水的试管中}
    \fact{自变量：}{台灯与试管的距离（10、20、30、40\,cm）}
    \fact{因变量：}{每分钟产生的气泡数}
    \fact{控制变量：}{水温、CO$_2$ 浓度、植物大小}
\end{notefacts}

% [GUIDE] The one allowed table: 2 things on 3 attributes, every cell short.
% [GUIDE] Adding a row such as "能量变化" whose cells need a sentence would

\notehead{光反应与暗反应对比}\src{2-3}
\begin{center}
\begin{tabular}{lll}
\toprule
 & 光反应 & 暗反应 \\
\midrule
场所 & 类囊体膜 & 叶绿体基质 \\
是否需要光 & 需要 & 不直接需要 \\
产物 & O$_2$、ATP、NADPH & 糖类 \\
\bottomrule
\end{tabular}
\end{center}

% [GUIDE] A key formula: display it, then say what the symbols mean.
\notehead{总反应式}\src{1}
\notelead{在叶绿体中、有光的条件下：}
\[
6\,\mathrm{CO_2} + 12\,\mathrm{H_2O} \xrightarrow[\text{叶绿体}]{\text{光}}
\mathrm{C_6H_{12}O_6} + 6\,\mathrm{O_2} + 6\,\mathrm{H_2O}
\]
\begin{itemize}
    \item 产物中 O$_2$ 的氧全部来自\key{水}，不来自 CO$_2$
    \item 水既是反应物又是产物：消耗 12 个，生成 6 个
\end{itemize}
\end{notesheet}
